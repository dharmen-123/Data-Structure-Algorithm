
        // // ****** VECTOR ****** //  //

// A vector in C++ STL is a dynamic array that can grow or shrink in size automatically. 
// Unlike normal arrays (fixed size), vectors manage memory internally and allow flexible operations.
// Think of a vector as a resizable array.

    // //  Key Features

// * Stores elements in contiguous memory (like arrays).
// * Dynamic resizing: Automatically expands/shrinks when elements are added/removed.
// * Provides random access (O(1) time to access any element).
// * Supports iterators for traversal.
// * Part of <vector> header in STL.

    // // Vector Implementation

// * Header File
#include <vector>

// *Declaration Syntax
vector<data_type> vector_name;
    Examples:
vector<int> v1;          // empty vector of integers
vector<float> v2;        // empty vector of floats
vector<string> v3;       // empty vector of strings

// *Initialization Syntax
vector<int> v = {10, 20, 30};   // initializer list
vector<int> v(5, 0);            // size 5, all elements = 0
vector<int> v2(v);              // copy constructor

// *Accessing Elements
v[i];        // direct access (like array)
v.at(i);     // safe access with bounds checking
v.front();   // first element
v.back();    // last element

// *Insertion and Deletion
v.push_back(40);   // insert at end
v.pop_back();      // remove last element
v.clear();         // remove all elements

// *Iterators
vector<int>::iterator it;
for(it = v.begin(); it != v.end(); it++) {
    cout << *it << " ";
}
